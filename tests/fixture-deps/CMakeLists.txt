cmake_minimum_required(VERSION 3.13)
project(fixture_deps LANGUAGES Fortran)
include(GNUInstallDirs)
add_subdirectory(src/third_party/PENF) # bundled in the release tarball (fobis fetch at release time)
add_library(fixture_deps STATIC src/lib/fixture_deps.f90)
target_link_libraries(fixture_deps PUBLIC PENF)
set_target_properties(fixture_deps PROPERTIES Fortran_MODULE_DIRECTORY ${CMAKE_BINARY_DIR}/mod)
install(TARGETS fixture_deps ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY ${CMAKE_BINARY_DIR}/mod/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
